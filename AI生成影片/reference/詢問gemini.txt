2. Python (Turtle) 自動繪畫程式碼
如果你希望用程式碼「看著電腦一步步把它畫出來」，可以使用 Python 標準庫自帶的 turtle：
import turtle

def draw_designer():
    screen = turtle.Screen()
    screen.setup(width=600, height=700)
    screen.title("簡易線條 - 青年室內設計師")
    screen.bgcolor("#F8F9FA")

    t = turtle.Turtle()
    t.speed(3)
    t.pensize(3)
    t.color("#2C3E50")

    # --- 1. 畫臉型輪廓 ---
    t.penup()
    t.goto(-50, 50)
    t.pendown()
    t.setheading(-60)
    t.circle(60, 60) # 左臉頰
    t.circle(30, 60) # 下巴
    t.circle(60, 60) # 右臉頰

    # --- 2. 畫高領與脖子 ---
    t.penup()
    t.goto(-30, -5)
    t.pendown()
    t.setheading(-90)
    t.forward(30) # 左脖
    
    t.penup()
    t.goto(30, -5)
    t.pendown()
    t.forward(30) # 右脖

    # 高領橫線
    t.penup()
    t.goto(-35, -35)
    t.pendown()
    t.setheading(0)
    t.forward(70)
    t.penup()
    t.goto(-35, -50)
    t.pendown()
    t.forward(70)

    # --- 3. 畫西裝肩膀與領片 ---
    # 左肩膀
    t.penup()
    t.goto(-35, -50)
    t.pendown()
    t.setheading(200)
    t.forward(100)
    
    # 右肩膀
    t.penup()
    t.goto(35, -50)
    t.pendown()
    t.setheading(-20)
    t.forward(100)

    # 西裝V領
    t.penup()
    t.goto(-20, -50)
    t.pendown()
    t.goto(0, -150)
    t.goto(20, -50)

    # --- 4. 畫五官（微笑與眼神）---
    # 左眼
    t.penup()
    t.goto(-30, 70)
    t.pendown()
    t.setheading(0)
    t.circle(8, 180)

    # 右眼
    t.penup()
    t.goto(10, 70)
    t.pendown()
    t.circle(8, 180)

    # 微笑
    t.penup()
    t.goto(-15, 25)
    t.pendown()
    t.setheading(-60)
    t.circle(18, 120)

    # --- 5. 蓬鬆髮型線條 ---
    t.penup()
    t.goto(-55, 80)
    t.pendown()
    t.setheading(80)
    t.circle(-60, 100) # 頭頂弧線
    t.circle(-40, 80)

    # 劉海造型
    t.penup()
    t.goto(0, 125)
    t.pendown()
    t.goto(-25, 85) # 中分左劉海
    t.penup()
    t.goto(0, 125)
    t.pendown()
    t.goto(25, 85)  # 中分右劉海

    t.hideturtle()
    screen.mainloop()

if __name__ == "__main__":
    draw_designer()
